# PyInstaller version-info resource for rapid-pdf.exe.
# Embeds product name / version / publisher so right-click -> Properties ->
# Details reads like a shipped app. Keep AppVersion in rapid-pdf.iss in sync.
#
# FileDescription IS THE NAME WINDOWS SHOWS. Explorer's "Open with" list, the
# Task Manager process row and the taskbar tooltip all read FileDescription,
# not ProductName. Up to 1.6.0 it held a whole tagline, so "Open with" offered
# a menu entry the width of the dialog. It is now just the app name, the same
# string as ProductName, because that is all any of those places have room for.
# The tagline lives where a tagline belongs: the About box, the installer's
# ApplicationDescription, and the landing site.
#
# THE NAME IS ONE WORD. "RapidPDF", not "Rapid PDF". Both fields carry the same
# string and tests/test_product_name.py asserts they agree, because they drifted
# once already and that is how the long one went unnoticed for four releases.
VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(1, 8, 1, 0),
    prodvers=(1, 8, 1, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo(
      [
        StringTable(
          '040904B0',
          [
            StringStruct('CompanyName', 'Lucas Ruiz'),
            StringStruct('FileDescription', 'RapidPDF'),
            StringStruct('FileVersion', '1.8.1.0'),
            StringStruct('InternalName', 'rapid-pdf'),
            StringStruct('LegalCopyright', 'Copyright (c) 2026 Lucas Ruiz'),
            StringStruct('OriginalFilename', 'rapid-pdf.exe'),
            StringStruct('ProductName', 'RapidPDF'),
            StringStruct('ProductVersion', '1.8.1.0')
          ]
        )
      ]
    ),
    VarFileInfo([VarStruct('Translation', [1033, 1200])])
  ]
)
